# Content pages are open to every crawler, AI crawlers included, and so are the public configuration
# and numbers that the pages fill in. Drives, links and pay pages are private; a drop page is public by design and
# carries its own noindex. A shop's page (/shop/<name>) may be read: a public shop asks to be indexed, every other
# one answers noindex (docs/shop.md "The address"), and public shops are listed in sitemap-shops.xml.

Allow: /
Allow: /api/v1/config$
Allow: /api/v1/stats$
Disallow: /s/
Disallow: /p/
Disallow: /drive
Disallow: /t$
Disallow: /api/
Disallow: /locker
Disallow: /inbox
Disallow: /send
Disallow: /watch
Disallow: /upgrade
Disallow: /verify
