/tmp, /var/tmp and /dev/shm executable, and that ExecPaths=/usr lifted ProtectSystem's read-only flag from /usr.

--- sandbox-probe.sh
#!/bin/bash
# sandbox-probe.sh <unit file> <probe command...>: run the probe as a throwaway unit with the unit's [Service]
# sandbox (everything but how it starts, restarts, logs and its runtime directory/credentials).
set -u
unit=$1; shift
props=()
while IFS= read -r line; do props+=(-p "$line"); done < <(
  awk '/^\[Service\]/{s=1; next} /^\[/{s=0} s' "$unit" | grep -E '^[A-Za-z]+=' |
  grep -vE '^(Type|ExecStart|ExecStartPre|ExecReload|Restart|RestartSec|TimeoutStopSec|TimeoutStartSec|SyslogIdentifier|KillSignal|StandardOutput|StandardError|RuntimeDirectory|RuntimeDirectoryMode|LoadCredential|EnvironmentFile)=')
systemd-run --wait --pipe --quiet --collect "${props[@]}" "$@"
--- view-check.sh
# Inside the sandbox: the visible (top-most) mount of each writable place is noexec, /usr is not a writable mount of
# its own, and exec from the private tmp places is refused.
awk '{top[$5] = $6} END {for (m in top) print m, top[m]}' /proc/self/mountinfo > /dev/shm/.m 2>/dev/null || true
bad=$(awk '{top[$5] = $6} END {for (m in top) if (top[m] ~ /(^|,)rw(,|$)/ && top[m] !~ /noexec/ && m !~ /^\/(proc|sys)(\/|$)/ && m != "/dev/hugepages") print m}' /proc/self/mountinfo | sort | tr '\n' ' ')
rm -f /dev/shm/.m
echo "visible writable mounts without noexec: ${bad:-none}"
awk '$5=="/usr" {print "/usr is a mount of its own: " $6}' /proc/self/mountinfo
for d in /tmp /var/tmp /dev/shm; do cp /usr/bin/true $d/.p 2>/dev/null && chmod 0700 $d/.p && { $d/.p 2>/dev/null && echo "$d: RAN" || echo "$d: refused"; rm -f $d/.p; }; done
