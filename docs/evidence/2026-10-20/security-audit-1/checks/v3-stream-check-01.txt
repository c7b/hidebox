# v3-stream independent check of server/mutis/indexer.py:parse_stream + Worker.mirror
import sys, os, json, logging, threading, tracemalloc
sys.path[:0] = ["/target/server", "/target/server/tests"]
os.chdir("/target/server/tests")
from http.server import ThreadingHTTPServer, BaseHTTPRequestHandler
from mutis import indexer, db, chain_worker
from mutis.indexer import Indexers, IndexerError
from mutis.fake_chain import fake_keys

logging.basicConfig(level=logging.INFO, stream=sys.stdout, format="LOG %(message)s")

SOS = b'{"sos":{"shard":1,"max_state_version":0,"has_more":false}}\n'
EOS = b'{"eos":{"max_state_version":1}}'
def payload(nonce_json): return SOS + b'{"update":{"Unspent":{"tag":1,"public_nonce":' + nonce_json + b'}}}\n' + EOS

print("== A. parse_stream unit behaviour ==")
for label, data in [("string nonce", payload(b'"deadbeef"')),
                    ("non-JSON line", SOS + b'garbage\n' + EOS),
                    ("8-digit int nonce 50000000", payload(b'50000000')),
                    ("10-digit int nonce 9999999999", payload(b'9999999999'))]:
    tracemalloc.start()
    try:
        r = indexer.parse_stream(data)
        cur, peak = tracemalloc.get_traced_memory()
        print(f"{label}: returned, nonce hex len={len(r[0]['unspent'][0][1])}, tracemalloc peak={peak}")
        del r
    except IndexerError as e:
        print(f"{label}: IndexerError {e.code}")
    except BaseException as e:
        print(f"{label}: raised {type(e).__name__}: {str(e)[:80]} (IndexerError? {isinstance(e, IndexerError)})")
    tracemalloc.stop()

print("== B. Worker.mirror with a hostile upstream ==")
class Hostile(BaseHTTPRequestHandler):
    hits = 0
    def do_POST(self):
        n = int(self.headers.get("Content-Length") or 0); self.rfile.read(n)
        Hostile.hits += 1
        body = payload(b'"deadbeef"')
        self.send_response(200); self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body))); self.end_headers(); self.wfile.write(body)
    def log_message(self, *a): pass
hostile = ThreadingHTTPServer(("127.0.0.1", 0), Hostile)
threading.Thread(target=hostile.serve_forever, daemon=True).start()
hurl = f"http://127.0.0.1:{hostile.server_address[1]}"

from harness import Stack
def rows(s): return s.conn.execute("SELECT COUNT(*) FROM chain_utxos").fetchone()[0]
for order in ("hostile-first", "hostile-last"):
    s = Stack(chain=True)
    try:
        w = s.worker
        live = s.indexer.url
        urls = [hurl, live] if order == "hostile-first" else [live, hurl]
        w.indexers = Indexers(urls, rate=1000)
        for _ in range(3):
            s.chain.add_output(fake_keys()["address"], 1000, None)
        print(f"-- {order}: upstreams={['hostile' if u == hurl else 'live' for u in urls]}")
        for tick in range(3):
            w.status.pop("mirror_ok_at", None) if tick == 0 else None
            w._step(w.mirror, s.now + 3 * tick)
            print(f"   tick {tick}: chain_utxos rows={rows(s)} mirror_ok_at={w.status.get('mirror_ok_at')} "
                  f"cursors={db.get_meta_json(s.conn, 'mirror_cursors', {})}")
        try:
            w.mirror(s.now + 20)
            print("   direct mirror(): returned")
        except BaseException as e:
            print(f"   direct mirror(): raised {type(e).__name__} (IndexerError? {isinstance(e, IndexerError)})")
    finally:
        s.close()
print("hostile hits:", Hostile.hits)
hostile.shutdown()
