check-01 (v3-caddysym): does the deployed Caddy 2.6.2 binary, with the release Caddyfile's @status route, follow a symlink planted in the status root?
Command (host): bash cf-audit/bin/sbx.sh v3-caddysym 120 -- bash /scratch/harness-01.sh > scratch/check-01.raw 2>&1
Sandbox: bwrap, no network (isolated loopback), empty env, single uid; the cross-uid part (drive plants, caddy reads its 0700 storage) is not reproducible here and rests on owner-observed modes + POSIX symlink semantics.
Inputs: dummy key file (not a real key) at Caddy's storage layout, mode 0600 in 0700 dirs; symlinks evidence-archive-x.json and evidence.json -> that file; evidence-archive-dir.json -> /.

===== harness-01.sh =====
#!/bin/bash
# v3-caddysym harness: deployed Caddy 2.6.2 binary + release Caddyfile (paths remapped), loopback only.
set -u
W=/work
export XDG_DATA_HOME=$W/caddyhome/.local/share XDG_CONFIG_HOME=$W/caddyhome/.config
echo "== caddy version"; /tools/caddy version; sha256sum /tools/caddy
mkdir -p $W/srv/mutis/status $W/srv/mutis/app/current/dist $W/mutis.d
echo '<!doctype html><title>404</title>' > $W/srv/mutis/app/current/dist/404.html
echo '{"ok":true}' > $W/srv/mutis/status/status.json
chmod 2775 $W/srv/mutis/status
# Dummy "caddy-only" private key at Caddy's certificate storage layout (certificates/<issuer>/<name>/<name>.key)
K=$XDG_DATA_HOME/caddy/certificates/acme-v02.api.letsencrypt.org-directory/dummy.test
mkdir -p $K; chmod -R 0700 $W/caddyhome
printf -- '-----BEGIN EC PRIVATE KEY-----\nDUMMY-NOT-A-REAL-KEY-v3-caddysym\n-----END EC PRIVATE KEY-----\n' > $K/dummy.test.key
chmod 0600 $K/dummy.test.key
# The "lower-trust writer" plants links in the status root (no access to the target needed to create a symlink)
ln -s $K/dummy.test.key $W/srv/mutis/status/evidence-archive-x.json
ln -s $K/dummy.test.key $W/srv/mutis/status/evidence.json
ln -s / $W/srv/mutis/status/evidence-archive-dir.json
echo "== status dir"; ls -la $W/srv/mutis/status; ls -la $K
# Release Caddyfile with only: admin off + auto_https off, /srv/mutis -> /work/srv/mutis,
# the import/redirect/catch-all blocks dropped, the site address pinned to loopback HTTP.
awk '(NR<38 || NR>47)' /target/deploy/caddy/Caddyfile | awk '/^# Any other Host name/{exit} {print}' | \
  sed -e 's#admin unix//run/caddy/admin.sock#admin off\n\tauto_https off\n\thttp_port 18080\n\thttps_port 18443#' \
    -e 's#/srv/mutis#/work/srv/mutis#g' \
    -e 's#^{\$MUTIS_HOST} {#http://127.0.0.1:18080 {#' > $W/Caddyfile
echo "== diff release Caddyfile vs test Caddyfile"; diff /target/deploy/caddy/Caddyfile $W/Caddyfile
/tools/caddy validate --config $W/Caddyfile --adapter caddyfile 2>&1 | tail -2
/tools/caddy run --config $W/Caddyfile --adapter caddyfile > $W/caddy.log 2>&1 &
P=$!
for i in $(seq 1 50); do curl -s -o /dev/null http://127.0.0.1:18080/status.json && break; sleep 0.2; done
for u in /status.json /evidence-archive-x.json /evidence.json /evidence-archive-dir.json; do
  echo "== GET $u"; curl -s -D - http://127.0.0.1:18080$u -o $W/body; echo "-- body:"; head -c 400 $W/body; echo
done
echo "== sha256 of served body for /evidence-archive-x.json vs key file"
curl -s http://127.0.0.1:18080/evidence-archive-x.json | sha256sum; sha256sum $K/dummy.test.key
kill $P; wait $P 2>/dev/null
echo "== caddy log (errors only)"; tail -5 $W/caddy.log
